// Payment Integration JavaScript
class PaymentIntegration {
    constructor(config) {
        this.gateway = config.gateway;
        this.endpoint = config.endpoint;
        this.apiKey = config.apiKey;
        this.metrics = {
            successCount: 0,
            failureCount: 0,
            totalAmount: 0
        };
    }

    async processPayment(amount, currency = 'USD') {
        const startTime = Date.now();
        
        try {
            const response = await fetch(this.endpoint + '/charges', {
                method: 'POST',
                headers: {
                    'Authorization': 'Bearer ' + this.apiKey,
                    'Content-Type': 'application/json'
                },
                body: JSON.stringify({
                    amount: Math.round(amount * 100),
                    currency: currency.toLowerCase()
                })
            });

            const result = await response.json();
            const duration = Date.now() - startTime;
            
            if (response.ok) {
                this.metrics.successCount++;
                this.metrics.totalAmount += amount;
                
                return {
                    success: true,
                    transactionId: result.id,
                    amount: amount,
                    currency: currency,
                    duration: duration + 'ms'
                };
            } else {
                throw new Error(result.error.message);
            }
        } catch (error) {
            this.metrics.failureCount++;
            return {
                success: false,
                error: error.message,
                duration: (Date.now() - startTime) + 'ms'
            };
        }
    }

    getMetrics() {
        const total = this.metrics.successCount + this.metrics.failureCount;
        const successRate = total > 0 ? (this.metrics.successCount / total * 100).toFixed(1) + '%' : '0%';
        
        return {
            ...this.metrics,
            successRate,
            totalTransactions: total
        };
    }
}

// Usage example
const paymentIntegration = new PaymentIntegration({
    gateway: 'Stripe',
    endpoint: 'https://api.stripe.com/v1',
    apiKey: 'sk_test_...'
});

paymentIntegration.processPayment(99.99, 'USD')
    .then(result => console.log('Payment result:', result))
    .catch(error => console.error('Payment error:', error));
